else if (command == "print")
{	
	string rest;
	getline(iss, rest);

	size_t start = rest.find_first_not_of(' ');
	if (start != string::npos)
	{
		rest = rest.substr(start);
	}
	
	if (rest.empty())
	{
		hasError = true;
		cerr << "Error on line " << currentLineNum << ": ";
		cerr << "Error: Missing argument for print statement" << endl;
		return;
	}
	else if(rest.size() > 1 && rest.front() == '"' && rest.back() == '"')
	{
		string literal = rest.substr(1, rest.size() - 2);

		//I used Zero for print size because it is a string literal and it is not allocated to a variable
		//I used zero fot the starting address because it is a string literal and it is allocated at runtime
		//because it doesnt have a variable it is allocated to
		int addr = allocateStringLiteral(literal, nextData, instructionMemory,hasError,0,0);
		instructionMemory[nextInstruction++] = 5100 + addr;
		return;
	}
	else
	{
		std::istringstream iss(rest);
		while (iss >> arg1)
		{
			if (st.type(arg1) == "string")
			{
				int adr = parseAddress(arg1, st);
				instructionMemory[nextInstruction++] = 5100 + adr;
				return;
			}
			int addr = parseAddress(arg1, st);

			if (addr == -1)
			{
				hasError = true;
				cerr << "Error on line " << currentLineNum << ": ";
				cerr << "\nError: Variable '" << arg1 << "' could not be Located" << endl;
				return;
			}
			instructionMemory[nextInstruction++] = 1100 + addr;
		}
